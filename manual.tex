\section*{软件说明书}

\subsection*{一、软件的功能描述}

Reading Share Chrome Extension（阅读分享Chrome扩展）是一款为Google Chrome浏览器设计的网页内容管理工具。本软件通过在浏览器工具栏集成快速收藏功能，帮助用户高效地收集、存储、管理和检索网络上发现的有价值内容。

\subsubsection*{主要功能模块}

\textbf{1. 网页收藏模块}
\begin{itemize}
  \item 一键收藏当前浏览网页的完整信息（标题、URL、收藏时间等）
  \item 自动提取网页元数据（网页标题、Meta描述、网站图标等）
  \item 支持编辑和更新已收藏项目的信息
  \item 支持批量删除收藏内容
  \item 为每个收藏项目添加备注和标签
\end{itemize}

\textbf{2. 内容搜索与过滤模块}
\begin{itemize}
  \item 实时关键词搜索功能，支持多字段搜索（标题、URL、备注等）
  \item 智能模糊匹配算法，支持中英文搜索
  \item 搜索结果实时更新和高亮显示
  \item 支持多条件组合搜索
  \item 搜索结果统计和展示
\end{itemize}

\textbf{3. 数据导入导出模块}
\begin{itemize}
  \item 从Chrome浏览器书签导入收藏
  \item 从HTML标准书签格式文件导入
  \item 导出收藏数据为JSON格式（用于数据分析和备份）
  \item 导出收藏数据为HTML格式（用于浏览器兼容和分享）
  \item 支持重复项目检测和处理
\end{itemize}

\textbf{4. 数据管理模块}
\begin{itemize}
  \item 本地IndexedDB数据库存储管理
  \item 数据永久保存（不受浏览器缓存清理影响）
  \item 数据隐私保护（所有数据存储在本地）
  \item 支持完整的数据导出备份
  \item 支持清空所有数据的操作
\end{itemize}

\textbf{5. 用户界面与交互模块}
\begin{itemize}
  \item 浏览器弹出窗口UI设计
  \item 列表视图和紧凑视图两种显示模式
  \item 浅色和深色主题支持
  \item 响应式设计，适应不同屏幕尺寸
  \item 虚拟滚动实现，支持大数据集高效渲染
\end{itemize}

\textbf{6. 性能优化模块}
\begin{itemize}
  \item 虚拟滚动技术实现高效的列表渲染
  \item 自动加载功能，支持渐进式加载大量数据
  \item 内存管理优化，确保长时间运行稳定性
  \item 搜索性能优化，支持快速全文检索
\end{itemize}

\subsection*{二、主要技术特点}

\subsubsection*{1. 架构特点}
\begin{itemize}
  \item \textbf{模块化架构}：采用模块化设计模式，将功能分为收藏、搜索、导入导出、存储等独立模块
  \item \textbf{后台脚本}：使用Chrome扩展后台脚本(background script)处理后台任务和事件监听
  \item \textbf{弹出窗口UI}：使用popup机制提供轻量级用户界面
  \item \textbf{内容脚本}：支持在网页上下文中运行脚本，提取网页信息
\end{itemize}

\subsubsection*{2. 数据存储技术}
\begin{itemize}
  \item \textbf{IndexedDB}：采用浏览器原生的IndexedDB API作为本地数据库
  \item \textbf{异步操作}：使用Promise和async/await实现异步数据库操作
  \item \textbf{事务管理}：支持数据库事务保证数据一致性
  \item \textbf{索引优化}：建立多个数据索引加快查询速度
  \item \textbf{增量存储}：支持大规模数据存储，最大可利用浏览器配额（通常50MB以上）
\end{itemize}

\subsubsection*{3. 搜索算法}
\begin{itemize}
  \item \textbf{全文搜索}：实现基于正则表达式和字符串匹配的全文搜索
  \item \textbf{智能分词}：支持中英文混合搜索和分词处理
  \item \textbf{模糊匹配}：采用编辑距离算法实现容错搜索
  \item \textbf{多字段检索}：支持在标题、URL、备注等多个字段同时搜索
  \item \textbf{实时索引}：维护搜索索引保证快速响应
\end{itemize}

\subsubsection*{4. 性能优化技术}
\begin{itemize}
  \item \textbf{虚拟滚动}：实现虚拟滚动（Virtual Scrolling）技术，只渲染可见区域的DOM元素
  \item \textbf{懒加载}：实现数据的懒加载和渐进式加载
  \item \textbf{内存管理}：优化JavaScript对象生命周期，防止内存泄漏
  \item \textbf{缓存机制}：实现搜索结果缓存减少重复查询
  \item \textbf{事件委托}：使用事件委托减少事件监听数量
  \item \textbf{防抖和节流}：在搜索和滚动等高频事件中使用防抖/节流
\end{itemize}

\subsubsection*{5. 用户界面技术}
\begin{itemize}
  \item \textbf{响应式设计}：使用CSS Media Queries实现响应式布局
  \item \textbf{主题系统}：实现动态主题切换，支持浅色和深色模式
  \item \textbf{DOM操作}：使用原生JavaScript进行高效的DOM操作
  \item \textbf{样式预处理}：使用CSS变量实现主题定制
\end{itemize}

\subsubsection*{6. 数据处理技术}
\begin{itemize}
  \item \textbf{JSON处理}：支持JSON序列化和反序列化
  \item \textbf{HTML解析}：实现HTML书签文件的解析和转换
  \item \textbf{元数据提取}：从网页HTML中提取meta标签等元数据
  \item \textbf{字符编码}：正确处理UTF-8及其他字符编码
  \item \textbf{文件导入导出}：使用File API实现浏览器级别的文件操作
\end{itemize}

\subsubsection*{7. 安全特点}
\begin{itemize}
  \item \textbf{隐私保护}：所有数据存储在本地，不经过任何远程服务器
  \item \textbf{权限最小化}：仅请求必要的Chrome扩展权限
  \item \textbf{内容安全}：实现内容安全策略(CSP)防止XSS攻击
  \item \textbf{数据隔离}：扩展数据与其他应用数据完全隔离
\end{itemize}

\subsubsection*{8. 兼容性与可靠性}
\begin{itemize}
  \item \textbf{浏览器兼容}：支持Chrome 90+版本
  \item \textbf{多系统支持}：兼容Windows、macOS、Linux等操作系统
  \item \textbf{错误处理}：完善的错误捕获和异常处理机制
  \item \textbf{可靠性}：支持大规模数据存储和检索的稳定性测试
\end{itemize}

\section{简介}

Reading Share 是一款为Chrome浏览器设计的生产力工具，帮助用户高效管理和分享阅读内容。无论您是研究员、学生、内容创作者还是普通网络用户，都可以使用本扩展快速保存、整理和搜索您在网络上发现的有价值的内容。

\subsection{主要特点}
\begin{itemize}
  \item \textbf{一键收藏}：在任何网页上快速保存链接、标题和元数据
  \item \textbf{智能搜索}：强大的全文搜索功能，快速定位已收藏的内容
  \item \textbf{批量导入}：支持从浏览器收藏夹导入大量书签数据
  \item \textbf{离线访问}：所有数据存储在本地，无需网络连接即可查看
  \item \textbf{数据同步}：通过导入导出功能实现多设备数据同步
  \item \textbf{隐私保护}：数据完全存储在本地，不上传到任何远程服务器
\end{itemize}

\subsection{适用人群}
\begin{itemize}
  \item \textbf{学生和研究人员}：收集研究资料、论文和参考文献

  \item \textbf{知识工作者}：整理行业新闻、技术文章和最佳实践

% ===== 2. 安装与启用 =====
\section{安装与启用}

\subsection{系统要求}
\begin{itemize}
  \item Chrome 浏览器版本 90 或更高
  \item 操作系统：Windows、macOS、Linux
  \item 磁盘空间：最少 10 MB
  \item 网络：仅用于导入收藏夹功能（可选）
\end{itemize}

\subsection{开发者模式安装（开发版）}
\begin{itemize}
  \stepitem{打开扩展管理页面}{%
    \begin{enumerate}
      \item 在 Chrome 地址栏输入 \texttt{chrome://extensions}
      \item 按 Enter 键访问扩展管理页面
      \item 或点击 Chrome 菜单 > 更多工具 > 扩展程序
    \end{enumerate}
  }
  \stepitem{启用开发者模式}{%
    \begin{enumerate}
      \item 找到页面右上角的"开发者模式"开关
      \item 点击开关启用开发者模式（开关变为蓝色）
      \item 此时页面会显示新的功能按钮：加载已解压的扩展程序、打包扩展程序等
    \end{enumerate}
  }
  \stepitem{加载扩展程序}{%
    \begin{enumerate}
      \item 点击"加载已解压的扩展程序"按钮
      \item 在弹出的文件选择对话框中，导航到项目根目录
      \item 选择项目文件夹（包含 manifest.json 的文件夹）
      \item 点击"选择文件夹"完成安装
    \end{enumerate}
  }
  \stepitem{验证安装}{%
    \begin{enumerate}
      \item 扩展管理页面会显示 "Reading Share Chrome Extension" 条目
      \item 条目显示为"启用"状态（有蓝色的启用开关）
      \item Chrome 工具栏右侧会出现 Reading Share 扩展图标
      \item 如未看到图标，点击 Chrome 工具栏右侧的扩展程序图标按钮，查找 Reading Share 并钉选到工具栏
    \end{enumerate}
  }
\end{itemize}

\subsection{卸载与重新安装}
\begin{itemize}
  \stepitem{临时禁用}{%
    在扩展管理页面中，关闭 Reading Share 条目右侧的蓝色启用开关，扩展会被禁用但不会删除数据
  }
  \stepitem{完全卸载}{%
    \begin{enumerate}
      \item 在扩展管理页面找到 Reading Share 条目
      \item 点击右下方的"删除"按钮
      \item 在确认对话框中再次点击"删除"
      \item \textit{警告：卸载前请导出数据，否则所有本地数据将被删除}
    \end{enumerate}
  }
  \stepitem{重新安装}{%
    如需重新安装，删除后按照上述步骤重新加载扩展程序即可
  }
\end{itemize}

% ===== 3. 快速开始 =====
\section{快速开始}

\subsection{首次使用步骤}
\begin{itemize}
  \stepitem{打开扩展弹窗}{%
    \begin{enumerate}
      \item 在 Chrome 工具栏找到 Reading Share 扩展图标（通常是书签或阅读相关的图标）
      \item 点击图标打开扩展的弹出窗口
      \item 弹出窗口会显示扩展的主界面
    \end{enumerate}
  }
  \stepitem{了解界面布局}{%
    第一次打开时，如果您还没有收藏任何内容，界面会显示一个欢迎提示和引导说明
  }
  \stepitem{收藏第一个网页}{%
    \begin{enumerate}
      \item 访问任何您喜欢的网站或网页
      \item 点击工具栏上的 Reading Share 扩展图标
      \item 当前网页的标题、URL 和其他信息会被自动收集
      \item 点击确认或提交按钮完成收藏
      \item 系统会显示"收藏成功"的提示消息
    \end{enumerate}
  }
\end{itemize}

\subsection{主要界面区域详解}

\subtitle{上方一键收藏网页}

\subtitle{基本收藏流程}
\begin{enumerate}
  \item \textbf{打开网页}：在 Chrome 中访问您想要收藏的网页
  \item \textbf{点击扩展图标}：点击 Chrome 工具栏上的 Reading Share 扩展图标
  \item \textbf{自动获取信息}：扩展会自动获取当前网页的以下信息：
    \begin{itemize}
      \item 网页标题（Page Title）
      \item 网页 URL（完整网址）
      \item 网页描述（如有 Meta Description 标签）
      \item 网页图标（Favicon）
      \item 当前时间（作为收藏时间戳）
    \end{itemize}
  \item \textbf{确认收藏}：检查自动填充的信息是否正确，点击确认或提交按钮
  \item \textbf{成功提示}：界面会显示"已添加到阅读收藏"或类似的成功提示消息
  \item \textbf{可选操作}：成功后可选择：
    \begin{itemize}
      \item 继续收藏其他网页
      \item 关闭扩展弹窗返回网页
      \item 打开收藏列表查看刚收藏的内容
    \end{itemize}
\end{enumerate}

\subtitle{编辑收藏信息}
\begin{itemize}
  \stepitem{打开收藏列表}{%
    点击工具栏的扩展图标打开弹出窗口
  }
  \stepitem{找到要编辑的项目}{%
    在列表中滚动查找或使用搜索功能定位要编辑的网页
  }
  \stepitem{点击编辑按钮}{%
    在列表项的右侧找到编辑按钮（通常是铅笔图标）并点击
  }
  \stepitem{修改信息}{%
    \begin{itemize}
      \item 修改网页标题：更改显示名称，便于识别
      \item 修改网址：如果 URL 有误，手动输入正确的网址
      \item 添加备注：在备注字段添加个人笔记或描述
      \item 修改标签：为这个收藏添加分类标签（如"技术"、"阅读"等）
    \end{itemize}
  }
  \stepitem{保存修改}{%
    点击保存或确认按钮，修改会立即生效
  }
\end{itemize}

\subtitle{删除收藏}
\begin{itemize}
  \item \textbf{单个删除}：
    \begin{enumerate}
      \item 在收藏列表中找到要删除的项目
      \item 点击该项目右侧的删除按钮（通常是垃圾桶图标）
      \item 系统会弹出确认对话框：``确定要删除此项目吗?''
      \item 点击确认即可删除，删除后无法撤销
    \end{enumerate}
  \item \textbf{批量删除}：
    \begin{enumerate}
      \item 在设置中可能提供清空所有数据的选项
      \item 点击"清空所有数据"按钮
      \item 系统会要求二次确认，输入 \texttt{DELETE} 或点击确认
      \item 所有收藏数据会被永久删除（建议先备份数据）
    \end{enumerate}
\end{itemize}

\subsection{强大的搜索功能}

\subtitle{基本搜索}
\begin{itemize}
  \stepitem{打开搜索框}{%
    打开扩展弹出窗口，搜索框位于顶部
  }
  \stepitem{输入关键词}{%
    在搜索框中输入您要查找的关键词
  }
  \stepitem{实时搜索}{%
    \begin{itemize}
      \item 大多数情况下，搜索是实时进行的
      \item 无需按 Enter 键，输入时已显示匹配结果
      \item 列表会自动过滤，仅显示匹配的项目
    \end{itemize}
  }
  \stepitem{查看结果}{%
    搜索结果在列表中高亮显示，点击任何结果可打开链接
  }
  \stepitem{清空搜索}{%
    点击搜索框右侧的清空按钮（X 图标）可清空搜索词，恢复显示所有内容
  }

\subtitle{方法一：通过弹出窗口}
\begin{enumerate}
  \item 点击 Chrome 工具栏上的 Reading Share 扩展图标
  \item 打开扩展弹出窗口
  \item 点击窗口右上角的设置按钮（齿轮图标）
  \item 打开设置面板或选项页面
\end{enumerate}

\subtitle{方法二：通过扩展管理页面}
\begin{enumerate}
  \item 打开 Chrome 扩展管理页面：\texttt{chrome://extensions}
  \item 找到 Reading Share 条目
  \item 点击"详情"或在条目下方找到"选项"链接
  \item 打开完整的设置页面
\end{enumerate}

\subtitle{方法三：右键菜单}
\begin{enumerate}
  \item 在 Chrome 工具栏上右键点击 Reading Share 扩展图标
  \item 在弹出的上下文菜单中选择"选项"
  \item 打开设置页面
\end{enumerate}
与隐私 =====
\section{数据管理与隐私}

\subsection{数据存储位置与方式}

\subtitle{本地存储}
\begin{itemize}
  \item \textbf{存储技术}：使用 Chrome 提供的 IndexedDB（浏览器本地数据库）
  \item \textbf{存储位置}：所有数据存储在您的计算机本地
  \item \textbf{存储位置详情}：
    \begin{itemize}
      \item Windows：\texttt{C:\textbackslash Users\textbackslash [用户名]\textbackslash AppData\textbackslash Local\textbackslash Google\textbackslash Chrome\textbackslash User Data\textbackslash Default\textbackslash IndexedDB}
      \item macOS：\texttt{\textasciitilde/Library/Application Support/Google/Chrome/Default/IndexedDB}
      \item Linux：\texttt{\textasciitilde/.config/google-chrome/Default/IndexedDB}
    \end{itemize}
  \item \textbf{访问权限}：仅有您的 Chrome 浏览器可访问这些数据
\end{itemize}

\subtitle{数据安全}
\begin{itemize}
  \item \textbf{不上传云端}：数据不会上传到 Reading Share 服务器或任何第三方服务
  \item \textbf{离线可用}：即使断网也可查看所有已收藏的内容
  \item \textbf{加密存储}：在某些系统上，浏览器会对本地存储进行加密
\end{itemize}

\subsection{数据持久化与删除}

\subtitle{何时数据会被保留}
\begin{itemize}
  \item 收藏后，数据会自动保存在本地数据库中
  \item 即使关闭浏览器或重启计算机，数据不会丢失
  \item 可以无限期保留数据（只要硬盘有空间）
\end{itemize}

\subtitle{何时数据会被删除}
\begin{itemize}
  \item \textbf{手动删除}：通过扩展界面删除单个项目
  \item \textbf{清空所有}：通过设置清空所有收藏数据
  \item \textbf{卸载扩展}：卸载扩展会删除相关的本地存储数据
  \item \textbf{清除浏览器缓存}：如果在清除浏览数据时选中"扩展程序"选项
  \item \textbf{重置 Chrome}：重置 Chrome 浏览器会删除所有扩展数据
\end{itemize}

\subtitle{清除浏览器缓存时的注意事项}
\begin{enumerate}
  \item 打开 Chrome 菜单 > 更多工具 > 清除浏览数据
  \item 或按 Ctrl+Shift+Delete（Windows）/ Cmd+Shift+Delete（Mac）
  \item 在打开的对话框中，确保 \textbf{不勾选} "扩展程序"复选框
  \item 否则 Reading Share 的所有数据会被清除
  \item 建议在清除前导出数据备份
\end{enumerate}

\subsection{安装和启用问题}

\subsubsection{Q：Chrome 扩展管理页面显示"扩展程序在此设备上不被允许"}

\begin{itemize}
  \item \textbf{原因}：某些企业 Chrome 设备或受管理的计算机禁用了非官方扩展
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 检查您是否在公司管理的 Chrome 帐户下
      \item 联系 IT 管理员获取权限
      \item 或在个人帐户下使用 Chrome
    \end{enumerate}
\end{itemize}

\subsubsection{Q：刷新后扩展图标消失了}

\begin{itemize}
  \item \textbf{原因}：浏览器更新可能重置了扩展的固定状态
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 打开 Chrome 扩展管理页面
      \item 确认 Reading Share 仍在列表中且状态为"启用"
      \item 点击工具栏右侧的扩展程序按钮
      \item 找到 Reading Share 并点击右侧的固定图标，重新钉选到工具栏
    \end{enumerate}
\end{itemize}

\subsubsection{Q：加载扩展时提示"无法加载来自该路径的扩展程序"}

\begin{itemize}
  \item \textbf{原因}：文件夹路径中可能包含特殊字符或中文字符
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 将项目文件夹移到路径更简单的位置
      \item 例如 \texttt{C:\textbackslash Projects\textbackslash reading-share-ext}
      \item 或重命名文件夹，使用英文和数字
      \item 重新加载扩展
    \end{enumerate}
\end{itemize}

\subsubsection{Q：manifest.json 校验失败}

\begin{itemize}
  \item \textbf{原因}：manifest.json 文件格式错误或损坏
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 用文本编辑器打开 manifest.json
      \item 检查 JSON 格式是否正确（所有引号、逗号等）
      \item 使用在线 JSON 验证工具检查：\url{https://jsonlint.com}
      \item 确保没有中文字符或特殊字符
      \item 修正格式后保存，重新加载扩展
    \end{enumerate}
\end{itemize}

\subsection{收藏与搜索问题}

\subsubsection{Q：收藏按钮无反应，无法收藏网页}

\begin{itemize}
  \item \textbf{原因可能}：
    \begin{enumerate}
      \item 扩展未正确加载或启用
      \item 当前网页是特殊网页（如 Chrome 内置页面）
      \item JavaScript 被禁用
      \item 浏览器缓存问题
    \end{enumerate}
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 确认扩展在 \texttt{chrome://extensions} 中状态为启用
      \item 尝试在普通网站（如 Wikipedia、GitHub 等）上收藏
      \item 刷新浏览器（Ctrl+R 或 Cmd+R）
      \item 重启 Chrome 浏览器
      \item 如果仍不行，卸载并重新安装扩展
    \end{enumerate}
\end{itemize}

\subsubsection{Q：收藏成功但列表中看不到新项目}

\begin{itemize}
  \item \textbf{原因}：界面未及时刷新或项目显示位置被搜索过滤
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 检查是否启用了搜索过滤，清空搜索框
      \item 点击清空按钮（X 图标）清除搜索条件
      \item 向上滚动列表查看最新的项目（新项目通常在最前面）
      \item 刷新扩展弹窗或重新打开扩展
      \item 检查浏览器控制台是否有错误（按 F12 打开开发者工具）
    \end{enumerate}
\end{itemize}

\subsubsection{Q：搜索结果为空或搜索很慢}

\begin{itemize}
  \item \textbf{原因}：
    \begin{enumerate}
      \item 搜索关键词不匹配任何内容
      \item 收藏数据库过大导致搜索变慢
      \item 浏览器性能问题
    \end{enumerate}
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 尝试更短的搜索关键词
      \item 清除搜索框，查看是否有收藏数据
      \item 检查是否无意中删除了所有数据
      \item 如果有数千个收藏，搜索可能需要几秒钟，这是正常的
      \item 关闭其他标签页和应用以释放系统资源
      \item 重启浏览器
    \end{enumerate}
\end{itemize}

\subsubsection{Q：无法打开搜索到的链接}

\begin{itemize}
  \item \textbf{原因}：原始网站可能离线、链接失效或被屏蔽
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 检查网址是否正确（编辑项目查看完整 URL）
      \item 尝试复制链接到地址栏手动访问
      \item 检查网络连接
      \item 查看收藏时填写的原始链接是否有错误
      \item 如果是失效链接，可编辑项目更新链接
      \item 使用 Wayback Machine (\url{https://web.archive.org}) 查看历史版本
    \end{enumerate}
\end{itemize}

  \item \textbf{每月}：进行完整的数据导出
  \item \textbf{每季度}：验证备份文件的完整性
\end{itemize}

\subsubsection{备份存储策略}
\begin{itemize}
  \item 本地存储：至少 2 个物理位置的副本
  \item 云存储：至少 1 个云服务备份（Google Drive、Dropbox 等）
  \item 离线存储：重要数据的光盘或 U 盘副本
\end{itemize}

\subsection{性能优化建议}

\subsubsection{管理收藏数量}
\begin{itemize}
  \item 定期清理陈旧数据保持库相对整洁
  \item 可以按年份分批管理（存档旧年份数据）
  \item 避免在单一数据库中保存超过 10 万条记录
\end{itemize}

\subsubsection{系统优化}
\begin{itemize}
  \item 关闭浏览器中不必要的扩展
  \item 定期清除浏览器缓存（但不清除扩展数据）
  \item 定期重启浏览器保持性能
  \item 更新 Chrome 到最新版本
\end{itemize}

\subsection{性能与兼容性问题}

\subsubsection{Q：扩展运行很慢，有数千个收藏时特别明显}

\begin{itemize}
  \item \textbf{原因}：收藏数据量大导致的性能问题
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 使用搜索功能缩小显示范围（减少渲染的项目数）
      \item 关闭浏览器其他标签页和应用释放资源
      \item 升级计算机 RAM（如果系统内存不足）
      \item 考虑定期清理过期或无用的收藏，保持数据库整洁
      \item 将大量数据分批导出，保持活跃数据库相对较小
    \end{enumerate}
\end{itemize}

\subsubsection{Q：滚动时出现卡顿或漂移}

\begin{itemize}
  \item \textbf{原因}：虚拟滚动实现问题或系统资源不足
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 尝试用虚拟滚动的替代模式（如果设置中有该选项）
      \item 关闭浏览器硬件加速（Chrome 菜单 > 设置 > 系统 > 关闭硬件加速）
      \item 清除浏览器缓存
      \item 禁用其他扩展来排除冲突
      \item 重启浏览器
    \end{enumerate}
\end{itemize}

\subsubsection{Q：与其他扩展冲突，出现异常行为}

\begin{itemize}
  \item \textbf{排查步骤}：
    \begin{enumerate}
      \item 进入 Chrome 扩展管理页面
      \item 逐个禁用其他扩展，每次禁用后测试 Reading Share
      \item 找出冲突的扩展
      \item 卸载冲突扩展，或联系扩展开发者反馈
    \end{enumerate}
  \item \textbf{常见冲突扩展}：
    \begin{itemize}
      \item 某些广告拦截器
      \item 页面修改类扩展
      \item 数据分析追踪类扩展
    \end{itemize}
\end{itemize}

\subsection{跨浏览器兼容性}

\subsubsection{Q：能在 Firefox、Edge 等浏览器中使用吗}

\begin{itemize}
  \item \textbf{当前状态}：Reading Share 目前仅为 Chrome 浏览器开发
  \item \textbf{解决方案}：
    \begin{enumerate}
      \item 考虑使用 Chromium 内核浏览器（Edge、Brave 等），某些版本可能兼容
      \item 在 Firefox 上，使用 Chrome 扩展兼容层（需要额外配置）
      \item 使用浏览器云同步功能（如 Chrome Sync）在设备间共享
      \item 通过导出/导入数据在不同浏览器间转移收藏
    \end{enumerate}永久删除所有收藏数据，无法撤销。确定要继续吗?''
  \item 某些版本要求输入确认码（如 \texttt{DELETE}）或再次点击确认按钮
  \item 点击确认后，所有数据会立即被删除
  \item 扩展会重启，显示空的收藏列表
\end{enumerate}

\subsection{导入收藏夹}

\subtitle{从浏览器导入 Chrome 收藏}
\begin{enumerate}
  \item \textbf{打开设置面板}：点击扩展弹出窗口右上角的设置按钮（齿轮图标）
  \item \textbf{找到导入选项}：在设置中找到"导入 Chrome 收藏夹"或"导入书签"选项
  \item \textbf{选择收藏夹}：
    \begin{itemize}
      \item 选择导入所有收藏夹，或
      \item 仅导入特定的书签文件夹
    \end{itemize}
  \item \textbf{确认导入}：点击开始导入按钮
  \item \textbf{进度显示}：导入过程中会显示进度条，显示已导入的数量
  \item \textbf{完成提示}：导入完成后显示"已导入 X 个书签"的成功提示
  \item \textbf{查看导入结果}：返回主界面，新导入的书签会显示在列表中
\end{enumerate}

\subtitle{从 HTML 文件导入}
\begin{enumerate}
  \item \textbf{准备 HTML 文件}：
    \begin{itemize}
      \item 从 Chrome 导出书签：Chrome 菜单 > 书签 > 书签管理器 > 右上方菜单 > 导出书签
      \item 或从其他浏览器导出为 HTML 格式的书签文件
    \end{itemize}
  \item \textbf{打开设置}：点击扩展设置按钮
  \item \textbf{选择导入文件}：找到"从文件导入"选项
  \item \textbf{选择 HTML 文件}：
    \begin{enumerate}
      \item 点击"选择文件"按钮
      \item 在系统文件对话框中定位书签 HTML 文件
      \item 选中后点击打开
    \end{enumerate}
  \item \textbf{开始导入}：系统自动解析 HTML 文件并导入书签数据
  \item \textbf{处理重复}：
    \begin{itemize}
      \item 如果导入的书签已存在，系统会提示如何处理
      \item 选项包括：跳过、覆盖、或保留两份副本
    \end{itemize}
  \item \textbf{导入完成}：显示导入统计信息和成功提示
\end{enumerate}

\subtitle{导入注意事项}
\begin{itemize}
  \item \textbf{备份原数据}：导入前请备份现有的收藏数据
  \item \textbf{检查导入结果}：导入后检查是否有重复或错误的条目
  \item \textbf{大量数据导入}：导入几千个书签时可能需要数分钟，请耐心等待
  \item \textbf{文件格式}：仅支持标准的 HTML 书签格式文件
\end{itemize}

\subsection{滚动加载优化}

\subtitle{虚拟滚动技术}

  \item \textbf{高效内存使用}：即使收藏了数千个网页，扩展仍然能保持快速响应
  \item \textbf{流畅滚动体验}：只在可视区域渲染列表项，滚动时不会卡顿
  \item \textbf{自动预加载}：当滚动接近底部时，自动加载下一批数据
\end{itemize}

\subtitle{使用滚动加载}
\begin{enumerate}
  \item \textbf{打开列表}：打开扩展弹出窗口显示收藏列表
  \item \textbf{向下滚动}：在列表中向下滚动查看更多内容
  \item \textbf{自动加载}：当滚动到列表接近底部时（通常最后几项可见时）
  \item \textbf{加载提示}：底部会显示"加载中..."或加载动画
  \item \textbf{继续浏览}：新加载的内容会自动添加到列表末尾
  \item \textbf{性能保证}：即使有数万个收藏项，滚动性能也很流畅
\end{enumeratize}
  \stepitem{收藏当前网页}{%
